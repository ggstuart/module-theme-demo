def average(marks):
    if not marks:
        return None
    return sum(marks) / len(marks)

print(average([72, 65, 80]))

SELECT s.name, AVG(m.mark) AS average
FROM students AS s
JOIN marks AS m ON m.student_id = s.id
GROUP BY s.name
ORDER BY average DESC;
